function findCharCounts(str) {
        let words = {};
        for (s of str) {
          if (Object.keys(words).includes(s)) {
            words[s] += 1;
          } else {
            words[s] = 1;
          }
        }
        return words;
      }
      console.log(findCharCounts(["a", "b", "c", "d", "a", "b"]));

function removeDuplicates(arr) {
        let t = [];
        for (let a of arr) {
          if (t.includes(a)) {
            continue;
          } else {
            t.push(a);
          }
        }
        return t;

4. let n = '' || "abc" || null;
console.log(n) // outputs abc 
why?
Ans:
In JavaScript, the logical OR (||) operator evaluates expressions from left to right and returns the first truthy value it encounters.

'' (Empty string): This is a falsy value. Because it is falsy, the || operator ignores it and moves to the next expression.
"abc" (Non-empty string): This is a truthy value. As soon as the operator finds a truthy value, it stops evaluating any further expressions (this is called short-circuit evaluation).
null: This value is never even looked at because the operation already found its result.

7. How to deep flatten an array in JavaScript?
 function deepFlatten(arr) {
        let result = [];
        for (let a of arr) {
          if (Array.isArray(a)) {
            result = result.concat(deepFlatten(a));
          } else {
            result.push(a);
          }
        }
        return result;
      }
      console.log(deepFlatten([10, 20, [30, 40], [[50, 60]], [[[77, 44]]]]));

9. WAP to find prime numbers between 20 to 40:
Ans:
  function generatePrimeNo(start, end) {
        let prime = [];
        for (let i = start; i <= end; i++) {
          let isPrime = true;

          if (i < 2) {
            isPrime = false;
          }
          for (let j = 2; j < i; j++) {
            if (i % j === 0) {
              isPrime = false;
              break;
            }
          }
          if (isPrime) {
            prime.push(i);
          }
        }
        return prime;
      }
      console.log("generatePrimeNo(20,40)", generatePrimeNo(20, 40));

13. WAP to sort without using sort()
Ans:
 function bubbleSort(arr) {
        for (let i = 0; i < arr.length; i++) {
          for (let j = 0; j < arr.length - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
              let t = arr[j];
              arr[j] = arr[j + 1];
              arr[j + 1] = t;
            }
          }
        }
        return arr;
      }

      console.log(bubbleSort([22, 33, 11, 54, 8,3]));

  14. What is recursion?
  Ans:
  Recursion in JavaScript is a programming technique where a function calls itself to solve a problem. 

  The Base Case: The exit condition that stops the function from calling itself.
  The Recursive Case: The part of the function where it calls itself, passing in a slightly modified or smaller input to move closer to the base case.
